longest_string.py Verified 
def longest(words):
    big = ""
    for word in words:
        if len(word) > len(big):
            big = word
    return big

# Demonstration Vector Execution
words = ["hi", "hello", "howdy", "hey"]
print(longest(words))  # Explicit Standard Output: howdy

def count_even_odd(numbers):
    even_count = 0
    odd_count = 0

    for num in numbers:
        if num % 2 == 0:
            even_count += 1
        else:
            odd_count += 1

    return even_count, odd_count

nums = [10, 3, 7, 8, 12, 5]
# Destructuring/Unpacking execution payload
even, odd = count_even_odd(nums)

print("Even numbers:", even)
print("Odd numbers:", odd)
